# 7.4. Quick Start

## Installation

```bash
npm install @triplef/core-logger
npm install @nestjs/common joi pino pino-pretty
```

## Basic Setup

### 1. Create a LoggerConfigService

```typescript
import { Injectable } from '@nestjs/common';
import { ValidateReturnValue } from '@triplef/config-factory/validate-return-value';
import { CoreLoggerSchema } from '@triplef/core-logger';

@Injectable()
export class LoggerConfigService {
  @ValidateReturnValue(CoreLoggerSchema)
  get options() {
    return {
      level: process.env.LOGGER_LEVEL || 'info',
      transport: {
        target: 'pino-pretty',
        options: {
          ignore: 'pid,hostname,context',
          messageFormat: (log, messageKey) =>
            log.context ? `[${log.context}] ${log[messageKey]}` : String(log[messageKey] ?? ''),
        },
      },
    };
  }
}
```

### 2. Register with CoreLoggerModule

```typescript
import { Module } from '@nestjs/common';
import { CoreLoggerModule } from '@triplef/core-logger';
import { LoggerConfigService } from './logger-config.service';

@Module({
  imports: [
    CoreLoggerModule.registerAsync({
      global: true,
      inject: [LoggerConfigService],
      useFactory: (config: LoggerConfigService) => config.options,
    }),
  ],
})
export class AppModule {}
```

### 3. Bootstrap

```typescript
import { NestFactory } from '@nestjs/core';
import { CoreLoggerService } from '@triplef/core-logger';
import { AppModule } from './app.module';

const app = await NestFactory.create(AppModule, { bufferLogs: true });
app.useLogger(app.get(CoreLoggerService));
```

### 4. Log with onLog

```typescript
logger.info('user signed up', {
  userId: '123',
  onLog: (entry) => sendToDb(entry),
});
```

## Import Paths

The library is tree-shakeable. Import only what you need:

```typescript
import { CoreLoggerModule, CoreLoggerSchema, CoreLoggerService } from '@triplef/core-logger';
import { CoreLoggerService } from '@triplef/core-logger/core-logger';
```
