# 9.4. Quick Start

## Installation

```bash
npm install @triplef/bullmq-logger
npm install @nestjs/common bullmq joi pino pino-pretty
```

## Basic Setup

### 1. Create a LoggerConfigService

```typescript
import { Injectable } from '@nestjs/common';
import { ValidateReturnValue } from '@triplef/config-factory/validate-return-value';
import { BullMQLoggerSchema } from '@triplef/bullmq-logger';

@Injectable()
export class LoggerConfigService {
  @ValidateReturnValue(BullMQLoggerSchema)
  get options() {
    return {
      level: process.env.BULLMQ_LOG_LEVEL || 'info',
      redact: {
        paths: ['apiKey', '*.apiKey', 'req.headers.authorization'],
        censor: '[REDACTED]',
      },
      transport: {
        target: 'pino-pretty',
        options: { ignore: 'pid,hostname' },
      },
    };
  }
}
```

### 2. Register with BullMQLoggerModule

```typescript
import { Module } from '@nestjs/common';
import { BullMQLoggerModule } from '@triplef/bullmq-logger';
import { LoggerConfigService } from './logger-config.service';

@Module({
  imports: [
    BullMQLoggerModule.registerAsync({
      global: true,
      inject: [LoggerConfigService],
      useFactory: (config: LoggerConfigService) => config.options,
    }),
  ],
})
export class AppModule {}
```

### 3. Log a job

```typescript
import { Injectable } from '@nestjs/common';
import { BullMQLoggerService } from '@triplef/bullmq-logger';

@Injectable()
export class MyProcessor {
  constructor(private readonly logger: BullMQLoggerService) {}

  async process(job: Job) {
    await this.logger.log(job);
    await this.logger.error(job);
  }
}
```

## Import Paths

The library is tree-shakeable. Import only what you need:

```typescript
import { BullMQLoggerModule, BullMQLoggerSchema, BullMQLoggerService } from '@triplef/bullmq-logger';
import { BullMQLoggerService } from '@triplef/bullmq-logger/bullmq-logger';
```
