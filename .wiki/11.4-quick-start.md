# 11.4. Quick Start

## Installation

```bash
npm install @triplef/ai-sdk
npm install @nestjs/common ai joi
```

## Basic Setup

### 1. Register with AiSdkModule

```typescript
import { Module } from '@nestjs/common';
import { AiSdkModule } from '@triplef/ai-sdk';
import { createOllama } from 'ollama-ai-provider-v2';

@Module({
  imports: [
    AiSdkModule.registerAsync({
      global: true,
      useFactory: () => ({
        streamChunkTimeoutMs: 60_000,
        streamTotalTimeoutMs: 600_000,
        generateTotalTimeoutMs: 300_000,
        enableSmoothStream: true,
        createModel: (name) =>
          createOllama({ baseURL: 'http://127.0.0.1:11434/api' })(name),
      }),
    }),
  ],
})
export class AppModule {}
```

### 2. Stream a chat

```typescript
import { Injectable } from '@nestjs/common';
import { AiSdkService } from '@triplef/ai-sdk';

@Injectable()
export class ChatService {
  constructor(private readonly aiSdk: AiSdkService) {}

  async stream(messages: InputMessage[]) {
    return this.aiSdk.streamChat({ model: 'llama3.2', messages });
  }
}
```

## Import Paths

The library is tree-shakeable. Import only what you need:

```typescript
import { AiSdkModule, AiSdkConfigSchema, AiSdkService } from '@triplef/ai-sdk';
import type { AiSdkConfig, InputMessage } from '@triplef/ai-sdk';
import { AiSdkModule } from '@triplef/ai-sdk/ai-sdk';
```
