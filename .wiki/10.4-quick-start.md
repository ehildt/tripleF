# 10.4. Quick Start

## Installation

```bash
npm install @triplef/socketio
npm install @nestjs/common fastify-socket.io joi socket.io
```

## Basic Setup

### 1. Create a SocketIOConfigService

```typescript
import { Injectable } from '@nestjs/common';
import { ValidateReturnValue } from '@triplef/config-factory/validate-return-value';
import { SocketIOConfigSchema } from '@triplef/socketio';

@Injectable()
export class SocketIOConfigService {
  @ValidateReturnValue(SocketIOConfigSchema)
  get options() {
    return {
      opts: {
        transports: ['websocket', 'polling'],
        cors: { origin: '*', credentials: true, methods: ['GET', 'POST'] },
      },
    };
  }
}
```

### 2. Register with SocketIOModule

```typescript
import { Module } from '@nestjs/common';
import { SocketIOModule } from '@triplef/socketio';
import { SocketIOConfigService } from './socket-io-config.service';

@Module({
  imports: [
    SocketIOModule.registerAsync({
      global: true,
      inject: [SocketIOConfigService],
      useFactory: (config: SocketIOConfigService) => config.options,
    }),
  ],
})
export class AppModule {}
```

### 3. Attach and emit

```typescript
// main.ts
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter, NestFastifyApplication } from '@nestjs/platform-fastify';
import { SocketIOModule } from '@triplef/socketio';

void (async () => {
  const app = await NestFactory.create<NestFastifyApplication>(AppModule, new FastifyAdapter());
  await SocketIOModule.attach(app);
  await app.listen({ port: 3000 });
})();
```

```typescript
// notifications.service.ts
import { Injectable } from '@nestjs/common';
import { SocketIOService } from '@triplef/socketio';

@Injectable()
export class NotificationsService {
  constructor(private readonly socketService: SocketIOService) {}

  notifyAll(notification: unknown) {
    this.socketService.emit('notification', notification);
  }
}
```

## Import Paths

The library is tree-shakeable. Import only what you need:

```typescript
import { SocketIOModule, SocketIOConfigSchema, SocketIOService } from '@triplef/socketio';
import type { SocketIOServerConfig } from '@triplef/socketio';
import { SocketIOModule } from '@triplef/socketio/socketio';
```
