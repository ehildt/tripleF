# 8.3. Quick Start

## Installation

```bash
npm install @triplef/bullmq
npm install @nestjs/bullmq @nestjs/common bullmq ioredis joi
```

## Basic Setup

### 1. Create a BullMQConfigService

```typescript
import { Injectable } from '@nestjs/common';
import { ValidateReturnValue } from '@triplef/config-factory/validate-return-value';
import { BullMQConfigSchema } from '@triplef/bullmq';

@Injectable()
export class BullMQConfigService {
  @ValidateReturnValue(BullMQConfigSchema)
  get options() {
    return {
      connection: { host: 'localhost', port: 6379 },
      defaultJobOptions: { attempts: 3, backoff: { type: 'exponential', delay: 1000 } },
    };
  }
}
```

### 2. Register with BullMQModule

```typescript
import { Module } from '@nestjs/common';
import { BullMQModule } from '@triplef/bullmq';
import { BullMQConfigService } from './bullmq-config.service';
import { MyProcessor } from './my.processor';

@Module({
  imports: [
    BullMQModule.registerAsync({
      global: true,
      inject: [BullMQConfigService],
      queues: ['harness', { name: 'vectorize', connection: { host: 'keydb' } }],
      processors: [MyProcessor],
      useFactory: (config: BullMQConfigService) => config.options,
    }),
  ],
})
export class AppModule {}
```

### 3. Inject a queue

```typescript
import { InjectQueue } from '@nestjs/bullmq';
import { Queue } from 'bullmq';

export class MyService {
  constructor(@InjectQueue('harness') private readonly queue: Queue) {}
}
```

## Import Paths

The library is tree-shakeable. Import only what you need:

```typescript
import { BullMQModule, BullMQConfigSchema } from '@triplef/bullmq';
import type { BullMQConfig, Queue } from '@triplef/bullmq';
import { BullMQModule } from '@triplef/bullmq/bullmq';
```
